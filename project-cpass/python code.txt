    )

    rfm['frequency'].rank(method='first'),
    q=3,
    labels=[1,2,3]
)

rfm['M'] = pd.qcut(
    rfm['monetary'].rank(method='first'),
    q=3,
    labels=[1,2,3]
)

rfm['RFM_Score'] = (
    rfm['R'].astype(str)

    + rfm['M'].astype(str)
)

def assign_segment(row):

    if row['R'] == 3 and row['F'] == 3 and row['M'] == 3:
        return 'Champions'

    elif row['F'] >= 2 and row['M'] >= 2:
        return 'Loyal Clients'

    elif row['R'] == 1:
        return 'At Risk'

    else:
        return 'Dormant'

rfm['segment'] = rfm.apply(
    assign_segment,
    axis=1
)

import pandas as pd

def classify_delivery_stage(status):
    status = str(status).lower()
    if status in ['submitted']:
        return 'submitted'
    elif status in ['processed']:
        return 'processed'
    elif status in ['sent']:
        return 'sent'
    elif status in ['delivered', 'answered']:
        return 'delivered'
    elif status in ['opened', 'read', 'clicked']:
        return 'engaged'
    elif status in ['failed', 'rejected', 'rna']:
        return 'failed'
    else:
        return 'unknown'

# Ensure 'delivery_stage' is created in df before merging into rfm_complete
df['delivery_stage'] = df['status'].apply(classify_delivery_stage)

# Re-create rfm_complete to include the 'delivery_stage' column
rfm_complete = rfm_final[['user_id', 'segment_x', 'Activity', 'Actionable']].merge(df, how='left', on='user_id')
rfm_complete = rfm_complete.merge(channel_stats, on='channel', how='left')

segment_channel_metrics = rfm_complete.groupby(['segment_x', 'channel']).apply(lambda g: pd.Series({
    'total_volume': g['volume'].sum(),
    'failed_volume': g.loc[g['delivery_stage'] == 'failed', 'volume'].sum(),
    'delivered_volume': g.loc[g['delivery_stage'].isin(['delivered', 'engaged', 'sent']), 'volume'].sum(),
    'engaged_volume': g.loc[g['delivery_stage'] == 'engaged', 'volume'].sum()
})).reset_index()

segment_channel_metrics['failure_rate'] = segment_channel_metrics['failed_volume'] / segment_channel_metrics['total_volume']
segment_channel_metrics['delivery_rate'] = segment_channel_metrics['delivered_volume'] / segment_channel_metrics['total_volume']
segment_channel_metrics['engagement_rate'] = segment_channel_metrics['engaged_volume'] / segment_channel_metrics['total_volume']

# Display the results
print(segment_channel_metrics[['segment_x', 'channel', 'failure_rate', 'delivery_rate', 'engagement_rate']])
